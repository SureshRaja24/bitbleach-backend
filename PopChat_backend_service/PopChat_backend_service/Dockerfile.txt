# Stage 1: Build
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src

# 1. Copy the entire repository into the container
COPY . .

# 2. Navigate to the folder containing the .sln and .csproj
# Based on your repo structure: Backend -> PopChat_backend_service
WORKDIR "/src/Backend/PopChat_backend_service"

# 3. Restore dependencies
RUN dotnet restore

# 4. Build and Publish
# We publish the specific project file to the /app folder
RUN dotnet publish "PopChat_backend_service/PopChat_backend_service.csproj" -c Release -o /app/publish

# Stage 2: Runtime
FROM mcr.microsoft.com/dotnet/aspnet:8.0
WORKDIR /app

# Copy the published files from the build stage
COPY --from=build /app/publish .

# Render environment setup
ENV ASPNETCORE_URLS=http://0.0.0.0:10000
EXPOSE 10000

# Start the application
ENTRYPOINT ["dotnet", "PopChat_backend_service.dll"]
